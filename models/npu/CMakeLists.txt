cmake_minimum_required(VERSION 3.20)
project(covenant-npu CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# The NPU driver's headers and our plain-C wrapper over them are vendored under
# vendor/, so this builds with no other checkout present. See PROVENANCE.md.
#
# The driver ships no link library on Windows, so tools/gen-xrt-implib.ps1
# reconstructs one from the installed driver itself into build/. Nothing binary
# is tracked.
set(XRT_INC  "${CMAKE_SOURCE_DIR}/vendor/xrt/include")
set(XRT_SHIM "${CMAKE_SOURCE_DIR}/vendor/xrt-shim")
set(XRT_LIB  "${CMAKE_SOURCE_DIR}/build/xrt_coreutil.lib"
     CACHE FILEPATH "XRT import lib, generated by tools/gen-xrt-implib.ps1")

if(NOT EXISTS "${XRT_LIB}")
    message(STATUS "${XRT_LIB} missing - run tools/gen-xrt-implib.ps1 first")
    message(STATUS "nothing to build without it; stopping cleanly")
    return()
endif()

add_executable(bench-dispatch
    tools/bench-dispatch.cpp
    "${XRT_SHIM}/xrt_shim.cpp"
)
target_include_directories(bench-dispatch PRIVATE "${XRT_SHIM}" "${XRT_INC}")

# The driver's headers only pick the standard-library type over the Boost one
# when the compiler reports its language version honestly, which MSVC does not
# do without this flag.
target_compile_options(bench-dispatch PRIVATE /Zc:__cplusplus)
target_link_libraries(bench-dispatch PRIVATE "${XRT_LIB}")
